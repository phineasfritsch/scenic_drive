import DesignSystem
import Handoff
import ScenicKit
import SwiftUI

struct DriveRows: View {
@Binding var selection: HandoffDrive

@State private var menus: [HandoffDrive: DriveMenu] = [:]

var body: some View {
VStack(alignment: .leading, spacing: 0) {
ForEach(DriveSelector.order.indices, id: \.self) { index in
if index > 0 {
Rectangle().fill(DesignTokens.border).frame(height: 1)
}
row(for: DriveSelector.order[index].drive)
}
}
.fixedSize(horizontal: false, vertical: true)
.task {
for entry in DriveSelector.order {
menus[entry.drive] = DriveRoute.menu(for: entry.drive)
}
}
.accessibilityElement(children: .contain)
.accessibilityIdentifier("home.driveRows")
}

private func row(for drive: HandoffDrive) -> some View {
let isSelected = selection == drive
let pick = menus[drive].map { $0.row(selected: nil) }
let fastest = menus[drive].flatMap { menu in menu.rows.first { $0.displayedMinutes == 0 } }
let via = pick.map { OptionVia.line(roads: $0.roads, fastest: fastest?.roads ?? []) } ?? ""
return Button {
selection = drive
} label: {
HStack(alignment: .center, spacing: 12) {
VStack(alignment: .leading, spacing: 2) {
Text(DriveCopy.shortName(for: drive))
.font(.headline)
.fontWeight(isSelected ? .bold : .semibold)
.foregroundStyle(DesignTokens.fg)
.fixedSize(horizontal: false, vertical: true)
if !via.isEmpty {
Text(via)
.font(.subheadline)
.foregroundStyle(DesignTokens.fgMuted)
.lineLimit(2)
.fixedSize(horizontal: false, vertical: true)
}
}
.frame(maxWidth: .infinity, alignment: .leading)
if let pick, pick.displayedMinutes > 0 {
VStack(alignment: .trailing, spacing: 0) {
Text("+\(pick.displayedMinutes)")
.font(.title2)
.fontWeight(.semibold)
.monospacedDigit()
.foregroundStyle(DesignTokens.fg)
Text("min")
.font(.caption)
.foregroundStyle(DesignTokens.fgMuted)
}
.fixedSize()
}
}
.padding(.vertical, 8)
.padding(.leading, 10)
.frame(maxWidth: .infinity, minHeight: 44, alignment: .leading)
.overlay(alignment: .leading) {
Capsule().fill(isSelected ? DesignTokens.primary : Color.clear).frame(width: 4)
}
.contentShape(Rectangle())
}
.buttonStyle(.plain)
.accessibilityAddTraits(isSelected ? [.isButton, .isSelected] : .isButton)
.accessibilityIdentifier("home.driveRow.\(drive.rawValue)")
}
}

struct GatedHandoffButton: View {
let isSafetyDisclaimerAcknowledged: Bool

let drive: HandoffDrive

let row: DriveMenuRow?

var place: Coordinate? = nil

let onBlocked: () -> Void

let onFailure: (String?) -> Void

private static let log = Logger(subsystem: "com.phineasfritsch.scenicdrive",
category: "GatedHandoffButton")

var body: some View {
Button {
attempt()
} label: {
Text(Copy.label)
.font(.headline)
.foregroundStyle(DesignTokens.onPrimary)
.frame(maxWidth: .infinity, minHeight: 44)
.background(
RoundedRectangle(cornerRadius: 16, style: .continuous)
.fill(DesignTokens.primary)
)
}
.buttonStyle(.plain)
.accessibilityIdentifier("home.openInAppleMaps")
}

func attempt() {
guard isSafetyDisclaimerAcknowledged else {
onBlocked()
return
}
do {
try SkylineHandoff.open(drive, row: row, place: place)
onFailure(nil)
} catch {
Self.log.error("handoff refused: \(String(describing: error), privacy: .public)")
onFailure(Copy.handoffFailed)
}
}

private enum Copy {
static let label = "Open in Apple Maps"

static let handoffFailed = "Couldn't open Apple Maps. Try again."
}
}
